    // V72.4.3.2.29 DIRECT_PACKED_CHROMA_SMOOTH
    //
    // The active 3840x2160 -> 640x360 path is exact633 and reads U/V directly
    // from NihAV's row-split PGMYUV payload. The V27/V28 planar U/V repair is
    // therefore not on this hot path.
    //
    // Sample a 7x7 chroma neighbourhood around each exact 3x3 footprint. This
    // crosses the rectangular Bink2 chroma block discontinuities seen in RAW
    // truth while preserving the clean luma plane untouched.
    private static int SampleV724329PackedChroma7x7(
        ReadOnlySpan<byte> payload,
        int yBytes,
        int sourceWidth,
        int sourceChromaWidth,
        int sourceChromaHeight,
        int centerX,
        int centerY,
        bool vPlane)
    {
        var startX = Math.Max(0, centerX - 3);
        var endX = Math.Min(sourceChromaWidth - 1, centerX + 3);
        var startY = Math.Max(0, centerY - 3);
        var endY = Math.Min(sourceChromaHeight - 1, centerY + 3);

        var planeOffset =
            vPlane
                ? sourceChromaWidth
                : 0;

        var sum = 0;
        var count = 0;

        for (var y = startY; y <= endY; y++)
        {
            var row =
                yBytes +
                y * sourceWidth +
                planeOffset;

            for (var x = startX; x <= endX; x++)
            {
                sum += payload[row + x];
                count++;
            }
        }

        return count > 0
            ? (sum + count / 2) / count
            : 128;
    }
