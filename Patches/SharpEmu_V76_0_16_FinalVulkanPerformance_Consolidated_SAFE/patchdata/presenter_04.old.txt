                return PresentModeKHR.FifoKhr;
            }

            // NVIDIA DLSS-G on Vulkan currently requires host VSync to be off.
            // Keep the saved UI setting intact and only change the runtime
            // present mode while frame generation is requested.
            var forceVsyncOffForDlssG = RequestedDlssFrameGeneration();
            if (!_videoOptions.VSync || forceVsyncOffForDlssG)
            {
                for (var index = 0u; index < modeCount; index++)
                {
                    if (modes[index] == PresentModeKHR.ImmediateKhr)
                    {
                        return PresentModeKHR.ImmediateKhr;
                    }
                }
            }

            for (var index = 0u; index < modeCount; index++)
            {
                if (modes[index] == PresentModeKHR.MailboxKhr)
                {
                    return PresentModeKHR.MailboxKhr;
                }
            }

            return PresentModeKHR.FifoKhr;
        }
