            if (instruction.Opcode.StartsWith(
                    "TBufferStoreFormat",
                    StringComparison.Ordinal))
            {
                // V76.0.7 decodes the complete GFX10 MTBUF store family and
                // carries its instruction FORMAT/D16 metadata, but typed store
                // packing/conversion is intentionally not approximated as raw
                // dwords. That inverse conversion is the next buffer step.
                error =
                    $"typed-buffer store conversion pending opcode={instruction.Opcode} " +
                    $"format={control.TypedFormat?.ToString() ?? "?"} d16={control.D16}";
                return false;
            }

            if (instruction.Opcode.StartsWith("BufferStoreDword", StringComparison.Ordinal) ||
                instruction.Opcode.StartsWith("BufferStoreFormat", StringComparison.Ordinal) ||
                instruction.Opcode.StartsWith("BufferStoreByte", StringComparison.Ordinal) ||
                instruction.Opcode.StartsWith("BufferStoreShort", StringComparison.Ordinal))
            {
                EmitExecConditional(() =>
                {
                    if (TryGetSubdwordStoreInfo(
                            instruction.Opcode,
                            out var byteCount,
                            out var sourceShift))
                    {
                        StoreBufferBytes(
                            bindingIndex,
                            byteAddress,
                            LoadV(control.VectorData),
                            byteCount,
                            sourceShift);
                        return;
                    }

                    for (uint index = 0; index < control.DwordCount; index++)
                    {
                        var address = index == 0
                            ? byteAddress
                            : IAdd(byteAddress, UInt(index * sizeof(uint)));
                        StoreGuestBufferDwordV74054(
                            bindingIndex,
                            address,
                            LoadV(control.VectorData + index));
                    }
                });

                return true;
            }
