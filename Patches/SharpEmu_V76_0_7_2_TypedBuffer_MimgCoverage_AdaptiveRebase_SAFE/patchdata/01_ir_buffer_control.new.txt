public sealed record Gen5BufferMemoryControl(
    uint DwordCount,
    uint VectorAddress,
    uint VectorData,
    uint ScalarResource,
    int OffsetBytes,
    bool IndexEnabled,
    bool OffsetEnabled,
    bool Glc,
    bool Slc,
    uint TypedComponentCount = 0,
    uint? TypedFormat = null,
    bool D16 = false) : Gen5InstructionControl
{
    // V76.0.7: MTBUF has a four-bit opcode on GFX10. The high opcode bit is
    // carried in the second dword and selects the packed-D16 family. Keep the
    // number of memory-format components separate from the number of 32-bit
    // VGPRs occupied by packed D16 data.
    public uint ComponentCount =>
        TypedComponentCount != 0 ? TypedComponentCount : DwordCount;
}
