            if (IsFormatBufferLoad(instruction.Opcode))
            {
                EmitBufferFormatLoad(
                    bindingIndex,
                    byteAddress,
                    control.ScalarResource,
                    control);
                return true;
            }

            for (uint index = 0; index < control.DwordCount; index++)
            {
                var address = index == 0
                    ? byteAddress
                    : IAdd(byteAddress, UInt(index * sizeof(uint)));
                StoreV(
                    control.VectorData + index,
                    LoadGuestBufferDwordV74054(bindingIndex, address));
            }

            return true;
        }

        private void EmitBufferFormatLoad(
            int bindingIndex,
            uint byteAddress,
            uint scalarResource,
            Gen5BufferMemoryControl control)
        {
            var descriptorWord3 = LoadS(scalarResource + 3);
            var typed = control.TypedFormat.HasValue;
            var unifiedFormat = typed
                ? UInt(control.TypedFormat.GetValueOrDefault())
                : BitwiseAnd(
                    ShiftRightLogical(descriptorWord3, UInt(12)),
                    UInt(0x7F));
            var (dataFormat, numberFormat) = DecodeGfx10BufferFormat(unifiedFormat);

            var canonical = new uint[4];
            for (var component = 0; component < canonical.Length; component++)
            {
                canonical[component] = LoadGfx10BufferFormatComponent(
                    bindingIndex,
                    byteAddress,
                    dataFormat,
                    numberFormat,
                    component);
            }

            // V76.0.7: MTBUF uses FORMAT from the instruction and identity
            // destination selection. MUBUF FORMAT uses the resource descriptor
            // format plus descriptor dst_sel. The old path applied descriptor
            // FORMAT/dst_sel to both families, which mistranslated every MTBUF.
            var output = new uint[4];
            if (typed)
            {
                for (var component = 0; component < output.Length; component++)
                {
                    output[component] = canonical[component];
                }
            }
            else
            {
                var one = Gfx10FormatOne(numberFormat);
                for (var destination = 0; destination < output.Length; destination++)
                {
                    var selector = BitwiseAnd(
                        ShiftRightLogical(
                            descriptorWord3,
                            UInt((uint)destination * 3)),
                        UInt(7));
                    var value = UInt(0);
                    value = SelectUInt(selector, 1, one, value);
                    value = SelectUInt(selector, 4, canonical[0], value);
                    value = SelectUInt(selector, 5, canonical[1], value);
                    value = SelectUInt(selector, 6, canonical[2], value);
                    value = SelectUInt(selector, 7, canonical[3], value);
                    output[destination] = value;
                }
            }

            if (!control.D16)
            {
                for (var component = 0;
                     component < checked((int)control.ComponentCount);
                     component++)
                {
                    StoreV(
                        control.VectorData + (uint)component,
                        output[component]);
                }

                return;
            }

            // GFX10 packs pairs of D16 format results into one VGPR. Floating
            // format results narrow from f32 to IEEE f16; integer formats keep
            // their low 16-bit integer representation.
            for (var component = 0;
                 component < checked((int)control.ComponentCount);
                 component += 2)
            {
                var low = NarrowGfx10FormatResultToD16(
                    output[component],
                    numberFormat);
                var high = component + 1 < checked((int)control.ComponentCount)
                    ? NarrowGfx10FormatResultToD16(
                        output[component + 1],
                        numberFormat)
                    : UInt(0);
                StoreV(
                    control.VectorData + (uint)(component / 2),
                    BitwiseOr(
                        BitwiseAnd(low, UInt(0xFFFF)),
                        ShiftLeftLogical(
                            BitwiseAnd(high, UInt(0xFFFF)),
                            UInt(16))));
            }
        }

        private uint NarrowGfx10FormatResultToD16(
            uint value,
            uint numberFormat)
        {
            var isUint = Equal(numberFormat, UInt(4));
            var isSint = Equal(numberFormat, UInt(5));
            var isInteger = _module.AddInstruction(
                SpirvOp.LogicalOr,
                _boolType,
                isUint,
                isSint);
            var integerBits = BitwiseAnd(value, UInt(0xFFFF));
            var floatBits = EmitFloatToHalf(value);
            return _module.AddInstruction(
                SpirvOp.Select,
                _uintType,
                isInteger,
                integerBits,
                floatBits);
        }
