        private uint NarrowGfx10FormatResultToD16(
            uint value,
            uint numberFormat)
        {
            // V76.0.8: Equal(value, constant) creates the SPIR-V constant
            // internally. Passing UInt(4/5) compared against the ID of the
            // constant rather than the numeric number-format value.
            var isUint = Equal(numberFormat, 4);
            var isSint = Equal(numberFormat, 5);
            var isInteger = _module.AddInstruction(
                SpirvOp.LogicalOr,
                _boolType,
                isUint,
                isSint);
            var integerBits = BitwiseAnd(value, UInt(0xFFFF));
            var floatBits = EmitFloatToHalf(value);
            return _module.AddInstruction(
                SpirvOp.Select,
                _uintType,
                isInteger,
                integerBits,
                floatBits);
        }
