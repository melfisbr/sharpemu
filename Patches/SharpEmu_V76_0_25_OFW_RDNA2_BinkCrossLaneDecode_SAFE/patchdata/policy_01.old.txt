    internal static bool AllowHostDecoder
    {
        get
        {
            if (string.Equals(
                    Environment.GetEnvironmentVariable("SHARPEMU_BINK_FORCE_GUEST"),
                    "1",
                    StringComparison.Ordinal))
            {
                return false;
            }

            // Explicit opt-out of hybrid.
            if (string.Equals(
                    Environment.GetEnvironmentVariable("SHARPEMU_BINK_HYBRID_HOST"),
                    "0",
                    StringComparison.Ordinal))
            {
                // Still honor classic opt-in.
                return string.Equals(
                    Environment.GetEnvironmentVariable("SHARPEMU_BINK_ALLOW_HOST_DECODER"),
                    "1",
                    StringComparison.Ordinal);
            }

            // Default ON: hybrid host decode for reliable intro/attract video.
            return true;
        }
    }
