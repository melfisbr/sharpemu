        private static bool IsSupportedDppControlV7617(uint control) =>
            control <= 0xFF ||
            control is >= 0x101 and <= 0x10F or
                >= 0x111 and <= 0x11F or
                >= 0x121 and <= 0x12F or
                >= 0x130 and <= 0x13F or
                0x140 or 0x141 or
                >= 0x150 and <= 0x15F or
                >= 0x160 and <= 0x16F or
                >= 0x1F0 and <= 0x1FF;
